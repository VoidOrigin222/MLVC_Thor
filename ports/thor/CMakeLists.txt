find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATH_SUFFIXES aarch64-linux-gnu)
find_library(TENSORRT_LIBRARY nvinfer REQUIRED)
find_path(FFMPEG_INCLUDE_DIR libavcodec/avcodec.h PATH_SUFFIXES aarch64-linux-gnu)
foreach(component avcodec avformat avutil swscale)
  # Development packages in a private sysroot contain static archives. Use
  # the matching installed shared runtime instead of pulling all codec deps.
  unset(FFMPEG_${component}_LIBRARY CACHE)
  find_library(FFMPEG_${component}_LIBRARY NAMES lib${component}.so.60 lib${component}.so.58 lib${component}.so.7 ${component}
    PATHS /usr/lib/aarch64-linux-gnu /lib/aarch64-linux-gnu NO_DEFAULT_PATH REQUIRED)
endforeach()
file(GLOB_RECURSE common_sources CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/common/src/*.cc")
list(FILTER common_sources EXCLUDE REGEX "/acl_runtime\\.cc$")
file(GLOB_RECURSE codec_sources CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/mlvc/src/*.cc")
list(FILTER codec_sources EXCLUDE REGEX "/(video_io|dvpp_jpeg_encoder|dvpp_jpeg_decoder|dvpp_h264_encoder|decode_prior_acl|encode_index_acl|fp16_yuv444_to_nv12_acl)\\.cc$")
file(GLOB_RECURSE app_sources CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/application/src/*.cc")
list(FILTER app_sources EXCLUDE REGEX "/runtime_environment\\.cc$")
add_library(mlvc_msrtc_rans STATIC ${PROJECT_SOURCE_DIR}/third_party/msrtc_rans/src/EntropyCoder.cpp)
target_include_directories(mlvc_msrtc_rans PUBLIC ${PROJECT_SOURCE_DIR}/third_party/msrtc_rans/include PRIVATE ${PROJECT_SOURCE_DIR}/third_party/msrtc_rans/private/include)
add_library(mlvc_thor STATIC ${common_sources} ${codec_sources} ${app_sources}
  tensorrt_runtime.cc tensor_engine.cc batch_stage.cc runtime_fallbacks.cc video_io.cc nvidia_video.cc nvdec_reader.cc nv12.cu)
target_compile_definitions(mlvc_thor PUBLIC MLVC_TENSORRT=1 MLVC_HAS_LIBAVFORMAT=1)
target_include_directories(mlvc_thor PUBLIC include ${PROJECT_SOURCE_DIR}/common/include ${PROJECT_SOURCE_DIR}/mlvc/include ${PROJECT_SOURCE_DIR}/application/include
 ${PROJECT_SOURCE_DIR}/third_party/tomlplusplus/include ${OpenCV_INCLUDE_DIRS} ${TENSORRT_INCLUDE_DIR} ${FFMPEG_INCLUDE_DIR})
target_link_libraries(mlvc_thor PUBLIC mlvc_msrtc_rans CUDA::cudart ${TENSORRT_LIBRARY} Threads::Threads ${OpenCV_LIBS}
 ${FFMPEG_avcodec_LIBRARY} ${FFMPEG_avformat_LIBRARY} ${FFMPEG_avutil_LIBRARY} ${FFMPEG_swscale_LIBRARY} ${CMAKE_DL_LIBS})
foreach(component gobject-2.0 glib-2.0 gstreamer-1.0 gstapp-1.0 gstbase-1.0 gstvideo-1.0)
 find_library(GST_${component} NAMES ${component} lib${component}.so.0 REQUIRED)
 target_link_libraries(mlvc_thor PUBLIC ${GST_${component}})
endforeach()
target_compile_options(mlvc_thor PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3;-march=armv8.2-a+fp16>)
foreach(app encode decode)
 add_executable(mlvc_${app}_thor ${PROJECT_SOURCE_DIR}/apps/${app}_main.cc)
 target_link_libraries(mlvc_${app}_thor PRIVATE mlvc_thor)
endforeach()
